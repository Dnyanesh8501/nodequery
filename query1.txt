psql -U postgres -d <dbname> -c "\copy (SELECT d.die_no, l.line_code, l.line_name, m.type,
 m.priority FROM (SELECT mdl.*, COUNT(*) OVER (PARTITION BY mdl.die_id, mdl.type, mdl.priority)
AS same_priority_lines FROM map_die_line mdl WHERE mdl.status = true) m JOIN
mst_die d ON d.die_id = m.die_id JOIN mst_line l ON l.line_id = m.line_id
WHERE m.same_priority_lines > 1 ORDER BY d.die_no, m.type, m.priority, l.line_code) 
TO '/home/snopadm/die_line_report_26-11-2026.csv' WITH CSV HEADER;"
